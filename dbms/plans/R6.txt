SELECT
    o.id AS order_id,
    o.table_number,
    o.total_amount,
    o.status,
    o.special_instructions,
    o.created_at,
    o.updated_at,
    oi.id AS order_item_id,
    oi.food_id,
    oi.food_name,
    oi.unit_price,
    oi.quantity,
    (oi.unit_price * oi.quantity) AS item_total,
    f.category AS current_food_category
FROM orders AS o
LEFT JOIN order_items AS oi
    ON o.id = oi.order_id
LEFT JOIN foods AS f
    ON oi.food_id = f.id
WHERE o.id = 1
